RUST · MEMORY.RS 
pub fn validate_memory_path(root: &Path, rel: &str)
    -> Result<PathBuf, MemoryError> {
    // reject absolute paths and parent jumps
    let p = Path::new(rel);
    if p.is_absolute() || p.components().any(|c| c == Component::ParentDir) {
        return Err(MemoryError::OutsideMemory);
    }
    Ok(root.join(p))
}
